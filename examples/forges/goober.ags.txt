<?def
(extends NPC)
(name "Goober")

(runInfo hurt onHit)
(runInfo countdown onTick)

function hurt(attacker, amount) {
  console.log("[Goober] " + amount + " damage?! health: " + this.health);
}

function countdown(dt) {
  // Victory warp (SS18.30): give the death cry and the topple their moment,
  // then the room of white awaits.
  if (!this._dead || this.warped) return;
  this.warpT = (this.warpT || 0) + dt;
  if (this.warpT >= 3) {
    this.warped = true;
    console.log("[Goober] you win. enjoy the void.");
    this.goToScene("Scene 3");
  }
}
